# Parent Class class LibraryItem def initialize(title, id) @title = title @id = id @available = true end def display_info puts "Title: #{@title}" puts "ID: #{@id}" puts "Available: #{@available}" end def borrow_item if @available @available = false puts "#{@title} has been borrowed." else puts "#{@title} is currently not available." end end def return_item @available = true puts "#{@title} has been returned." end end # Child Class (Inheritance) class Book < LibraryItem def initialize(title, id, author) super(title, id) # Calls parent constructor @author = author end # Method Overriding def display_info puts "Book Title: #{@title}" puts "Author: #{@author}" puts "Book ID: #{@id}" puts "Available: #{@available}" end end # Creating Objects book1 = Book.new("Ruby Programming", 101, "Matz") book2 = Book.new("Data Structures", 102, "Mark Allen") puts "---- Book Details ----" book1.display_info puts "\n---- Borrow Book ----" book1.borrow_item puts "\n---- Try Borrowing Again ----" book1.borrow_item puts "\n---- Return Book ----" book1.return_item puts "\n---- Updated Details ----" book1.display_info
